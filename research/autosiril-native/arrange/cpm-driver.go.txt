func main(){
 root:=flag.String("root","..","repository root")
 source:=flag.String("source","","composed CP/M source")
 upstream:=flag.String("upstream","","completed common CP/M flatten root")
 engineChoice:=flag.String("engine","","optional engine filter for native mutants")
 input:=flag.String("input","","independent arrangement matrix")
 out:=flag.String("out","","output directory")
 flag.Parse();check(os.MkdirAll(*out,0755))
 fixtures:=filepath.Join(*root,"research/autosiril-native/midi/fixtures")
 buildOne:=func()*hir.Module{h,e:=build(read(*source),read(filepath.Join(fixtures,"music.lanz")),read(filepath.Join(fixtures,"music-manifest.json")),filepath.Join(*root,"stdlib"));check(e);return h}
 raw:=hir.LowerModule(buildOne());check(mir2.Verify(raw))
 opts:=pipeline.DefaultOptions();opts.AssertMode="none";opts.DisableVariantSearch=true
 steps,e:=pipeline.CompileHIRSteps(buildOne(),opts);check(e)
 assembly:="ORG 0x100\nLD SP,0xeffe\nCALL disk_main\nHALT\n"+steps.Assembly
 a,e:=z80asm.NewAssembler().AssembleString(assembly);check(e);if len(a.Errors)!=0{panic(a.Errors)}
 if origin+len(a.Binary)+64>=floor{panic("CP/M image overlaps stack guard")}
 for name,b:=range map[string][]byte{"program.a80":[]byte(assembly),"program.com":a.Binary,"raw.mir":[]byte(raw.Dump()),"final.mir":[]byte(steps.MIR2Module.Dump())}{check(os.WriteFile(filepath.Join(*out,name),b,0644))}
 var matrix struct{Cases []arrangeCase `json:"cases"`};check(json.Unmarshal(read(*input),&matrix));if len(matrix.Cases)==0{panic("empty matrix")}
 rows:=[]map[string]any{}
 for _,tc:=range matrix.Cases{
  cfg:=tc.Config.bytes()
  for index,engine:=range []string{"raw-mir2","final-mir2","z80"}{
   if *engineChoice!=""&&*engineChoice!=engine{continue}
   dir:=filepath.Join(*out,tc.Name,engine);check(os.MkdirAll(dir,0755))
   inputWire:=tc.Wire
   if tc.CPMFile!=""{inputWire=read(filepath.Join(*upstream,tc.CPMFile));if !bytes.Equal(inputWire,tc.Wire){panic("native CP/M second-stage file differs")}}
   check(os.WriteFile(filepath.Join(dir,"EVENTS.BIN"),inputWire,0644))
   if tc.ExistingOutput!=nil{check(os.WriteFile(filepath.Join(dir,"ARRANGE.BIN"),tc.ExistingOutput,0644))}
   d:=&diskOS{dir:dir,handles:map[uint16]*os.File{},calls:map[byte]int{},failFunction:tc.OSFault.Function,failOccurrence:tc.OSFault.Occurrence}
   var get func(string,int,int)[]byte;var returned uint16
   cycles,stack,poolWrites:=0,0,0
   if index<2{
    m:=raw;if index==1{m=steps.MIR2Module}
    v:=mir2.NewVM(m);v.MaxMemory=65536;v.MaxSteps=1000000000
    get=func(name string,at,n int)[]byte{return v.ReadHeap(v.GlobalAddr(name)+int64(at),n)}
    for name,b:=range cfg{for j,value:=range b{v.WriteHeap(v.GlobalAddr(name)+int64(j),value)}}
    d.read=v.ReadHeap;d.write=v.WriteHeap
    closed,opened:=false,false;pool:=v.GlobalAddr("orn_bytes");inputOpen:=v.GlobalAddr("input_open")
    v.ObserveDataAccess=func(a mir2.VMDataAccess){if a.Op!=mir2.OpStore{return};if a.Addr==inputOpen{if a.Value==1{opened=true};if opened&&a.Value==0{closed=true}};if closed&&a.Addr>=pool&&a.Addr<pool+4096{poolWrites++};for name,b:=range cfg{at:=v.GlobalAddr(name);if a.Addr>=at&&a.Addr<at+int64(len(b)){panic("VM mutated arrangement config")}}}
    host:=func(args []mir2.Value)([]mir2.Value,error){return []mir2.Value{{I:int64(d.invoke(byte(args[0].I),uint16(args[1].I)))}},nil}
    v.Hosts["bdos8"]=host;v.Hosts["@bdos8"]=host
    r,e:=v.Call("disk_main",nil);check(e);if len(r)!=1{panic("invalid CP/M VM return")};returned=uint16(r[0].I)
   }else{
    z:=emulator.NewRemogattoZ80WithScreen();z.SetROMEnd(0)
    mem:=z.Memory();for i:=range mem{mem[i]=0xa5};copy(mem[origin:],a.Binary)
    for name,b:=range cfg{copy(mem[a.Symbols[name]:],b)}
    d.read=func(at int64,n int)[]byte{return mem[int(at):int(at)+n]};d.write=func(at int64,b byte){z.WriteMemory(uint16(at),b)}
    get=func(name string,off,n int)[]byte{at:=a.Symbols[name]+off;return mem[at:at+n]}
    mem[initialSP]=byte(returnPC&255);mem[initialSP+1]=byte(returnPC>>8)
    z.SetRegisters(emulator.Registers{PC:uint16(a.Symbols["disk_main"]),SP:initialSP})
    p:=emulator.NewProfiler();z.SetProfiler(p);minSP:=uint16(initialSP)
    opened,closed:=false,false;var openWrites uint32;poolAt:=a.Symbols["orn_bytes"];closeWrites:=make([]uint32,4096)
    for count:=0;z.GetPC()!=returnPC;count++{
     if count>=250000000{panic("native CP/M instruction limit")}
     pc:=z.GetPC()
     if pc!=5&&(int(pc)<origin||int(pc)>=origin+len(a.Binary)){panic("native CP/M PC outside image/BDOS")}
     if pc==5{r:=z.GetRegisters();r.A=d.invoke(byte(r.BC),r.DE);r.HL=0;r.PC=binary.LittleEndian.Uint16(mem[r.SP:r.SP+2]);r.SP+=2;z.SetRegisters(r)}else{z.Step()}
     now:=p.WriteCount[a.Symbols["input_open"]]
     if now!=openWrites{value:=mem[a.Symbols["input_open"]];if value==1{opened=true};if opened&&value==0&&!closed{closed=true;copy(closeWrites,p.WriteCount[poolAt:poolAt+4096])};openWrites=now}
     if z.GetSP()<minSP{minSP=z.GetSP()};if z.GetSP()<floor{panic("native CP/M stack floor")}
    }
    if z.GetSP()!=initialSP+2||opened&&!closed{panic("native CP/M return/lease")}
    returned=z.GetRegisters().HL;cycles=z.GetCycles();stack=int(initialSP-minSP)
    for j,w:=range p.WriteCount[poolAt:poolAt+4096]{poolWrites+=int(w-closeWrites[j])}
    if poolWrites!=0{panic("native CP/M wrote ornament pool after input close")}
    allowed:=make([]bool,65536)
    for _,g:=range steps.MIR2Module.Globals{at,ok:=a.Symbols[g.Name];if !ok||g.At!=nil||at<origin||at+mir2.ByteWidth(g.Ty)>origin+len(a.Binary){panic("invalid resident CP/M global")};for j:=0;j<mir2.ByteWidth(g.Ty);j++{allowed[at+j]=true}}
    for _,s:=range backendSpills(steps.Assembly,a.Symbols,a.Binary){for j:=0;j<s.Bytes;j++{allowed[s.Address+j]=true}}
    for at:=floor;at<initialSP;at++{allowed[at]=true}
    for at,w:=range p.WriteCount{if w!=0&&!allowed[at]{panic(fmt.Sprintf("native CP/M write outside storage%04x",at))};if allowed[at]&&at<floor&&p.ExecCount[at]!=0{panic("native CP/M executed data")}}
    for name,b:=range cfg{for j:=range b{if p.WriteCount[a.Symbols[name]+j]!=0{panic("native CP/M mutated configuration")}}}
    if len(p.IORead)!=0||len(p.IOWrite)!=0{panic("native CP/M unexpected port IO")}
    for _,span:=range [][2]int{{origin-64,origin},{origin+len(a.Binary),origin+len(a.Binary)+64},{floor-64,floor}}{for at:=span[0];at<span[1];at++{if mem[at]!=0xa5{panic("native CP/M canary")}}}
   }
   if tc.OSFault.Function!=0&&!d.injected{panic("CP/M fault injection not reached")}
   if len(d.handles)!=0{d.close();panic("CP/M target leaked FCB")}
   word:=func(name string)uint16{return binary.LittleEndian.Uint16(get(name,0,2))}
   if returned!=tc.Fault||word("fault")!=tc.Fault{panic(fmt.Sprintf("%s/%s fault%d return%d want%d",tc.Name,engine,word("fault"),returned,tc.Fault))}
   if get("group_open",0,1)[0]!=0||get("input_open",0,1)[0]!=0||get("output_open",0,1)[0]!=0||poolWrites!=0{panic("CP/M stream lease not closed")}
   if word("orn_count")>64||word("orn_used")>4096||word("group_members")>512{panic("CP/M capacity escaped")}
   for name,b:=range cfg{if !bytes.Equal(get(name,0,len(b)),b){panic("CP/M configuration changed")}}
   for _,item:=range []struct{Name string;Want *uint16}{{"lnote_count",tc.LNotes},{"orn_count",tc.Ornaments},{"orn_used",tc.OrnBytes}}{if item.Want!=nil&&word(item.Name)!=*item.Want{panic(fmt.Sprintf("%s/%s %s differs",tc.Name,engine,item.Name))}}
   actual,e:=os.ReadFile(filepath.Join(dir,"ARRANGE.BIN"));exists:=e==nil
   if e!=nil&&!os.IsNotExist(e){check(e)}
   if tc.Fault==0||tc.CompleteOutput{
    if !exists||len(actual)!=(len(tc.Expected)+127)/128*128||!bytes.Equal(actual[:len(tc.Expected)],tc.Expected){panic("CP/M logical output differs")}
    for _,b:=range actual[len(tc.Expected):]{if b!=0{panic("CP/M output padding differs")}}
   }
   if len(actual)!=0&&len(tc.Expected)!=0&&tc.ExistingOutput==nil{
    prefix:=append([]byte(nil),tc.Expected...);prefix=append(prefix,make([]byte,(-len(prefix))&127)...)
    if len(actual)>len(prefix)||!bytes.Equal(actual,prefix[:len(actual)]){panic("CP/M partial output prefix differs")}
   }
   if tc.ExistingOutput!=nil&&!bytes.Equal(actual,tc.ExistingOutput){panic("preexisting CP/M output was changed")}
   if tc.OutputExists!=nil&&exists!=*tc.OutputExists{panic("CP/M output visibility differs")}
   if tc.WrittenRecords!=nil&&len(actual)!=*tc.WrittenRecords*128{panic("CP/M partial output length differs")}
   if !bytes.Equal(read(filepath.Join(dir,"EVENTS.BIN")),tc.Wire){panic("CP/M input file changed")}
   row:=map[string]any{"case":tc.Name,"engine":engine,"fault":word("fault"),"lnote_count":word("lnote_count"),"ornament_count":word("orn_count"),"ornament_bytes":word("orn_used"),"pool_writes_after_input_close":poolWrites,"output_exists":exists,"physical_bytes":len(actual),"physical_sha256":hash(actual),"input_file_sha256":hash(inputWire),"native_input_file":tc.CPMFile,"bdos_calls":d.calls,"injected_failure":d.injected,"t_states":cycles,"stack_below_entry":stack}
   rows=append(rows,row);fmt.Printf("PASS %s/%s fault%d LNotes%d ornaments%d/%dB output%v/%dB\n",tc.Name,engine,word("fault"),word("lnote_count"),word("orn_count"),word("orn_used"),exists,len(actual))
  }
 }
 globals,spills:=0,0;for _,g:=range steps.MIR2Module.Globals{globals+=mir2.ByteWidth(g.Ty)};for _,s:=range backendSpills(steps.Assembly,a.Symbols,a.Binary){spills+=s.Bytes}
 b,e:=json.MarshalIndent(map[string]any{"protocol":"autosiril-arrange-cpm/v1","source_sha256":hash(read(*source)),"matrix_sha256":hash(read(*input)),"target_executions":len(rows),"cases":rows,"binary_sha256":hash(a.Binary),"image_bytes":len(a.Binary),"loaded_globals_bytes":globals,"static_spill_bytes":spills,"code_and_bootstrap_bytes":len(a.Binary)-globals-spills,"scope":"native CALL5 API traps with fixed stack/HALT verifier contract; not physical CP/M TPA/CCP bootstrap compatibility"},"","  ");check(e);check(os.WriteFile(filepath.Join(*out,"receipt.json"),append(b,'\n'),0644))
}
