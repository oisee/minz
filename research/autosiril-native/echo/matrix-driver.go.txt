type echoConfig struct {
 First uint16 `json:"first"`
 Second uint16 `json:"second"`
 Wide [3]bool `json:"wide"`
 Unvoiced [3]bool `json:"unvoiced"`
 Modes [3]byte `json:"modes"`
}
func(c echoConfig) bytes()map[string][]byte{
 first,second:=make([]byte,2),make([]byte,2);binary.LittleEndian.PutUint16(first,c.First);binary.LittleEndian.PutUint16(second,c.Second)

 return map[string][]byte{"delay_one":first,"delay_two":second,"config_wide":wide,"config_unvoiced":u,"source_modes":c.Modes[:]}
}

func linearData(d *formats.TRDFile,pos int)[]byte {t:=pos/16;return d.ReadSector(t%80,t/80,pos%16)}
func putLinearData(d *formats.TRDFile,pos int,b []byte){t:=pos/16;d.WriteSector(t%80,t/80,pos%16,b)}
func linearOffset(pos int)int {t:=pos/16;return ((t%80*2+t/80)*16+pos%16)*256}
func seedEchoDisk(notes,wire []byte)*formats.TRDFile {
 d:=&formats.TRDFile{};pos:=16
 for index,data:=range [][]byte{notes,wire}{
  if len(data)>65280{panic("fixture input exceeds TRD capacity")}
  slot:=index*16;name:="NOTES   N";if index==1{name="ARRANGE A"};copy(d.Data[slot:slot+9],name)
  sectors:=(len(data)+255)/256;binary.LittleEndian.PutUint16(d.Data[slot+11:slot+13],uint16(len(data)))
  d.Data[slot+13]=byte(sectors);d.Data[slot+14]=byte(pos%16);d.Data[slot+15]=byte(pos/16)
  for i:=0;i<sectors;i++{b:=make([]byte,256);end:=(i+1)*256;if end>len(data){end=len(data)};copy(b,data[i*256:end]);putLinearData(d,pos+i,b)};pos+=sectors
 }
 sys:=linearData(d,8);sys[225]=byte(pos%16);sys[226]=byte(pos/16);sys[227]=22;sys[228]=2;binary.LittleEndian.PutUint16(sys[229:231],uint16(2560-pos));putLinearData(d,8,sys);return d
}

func configureEchoDisk(d *formats.TRDFile,control string){
 if control==""{return};sys:=linearData(d,8);count:=int(sys[228]);slot:=0
 if strings.HasPrefix(control,"arrange:"){slot=16;control=strings.TrimPrefix(control,"arrange:")}else{control=strings.TrimPrefix(control,"notes:")}
 switch control{
 case "missing_input":copy(d.Data[slot:slot+8],"OTHER   ")
 case "existing_output":copy(d.Data[count*16:count*16+9],"DELAYED D");sys[228]++
 case "duplicate_input":copy(d.Data[count*16:count*16+16],d.Data[slot:slot+16]);sys[228]++
 case "bad_geometry":sys[225]=16
 case "bad_directory_count":sys[228]=129
 case "directory_count_mismatch":sys[228]++
 case "bad_source_extent":d.Data[slot+13]=0
 case "source_length_under":binary.LittleEndian.PutUint16(d.Data[slot+11:slot+13],13)
 case "source_length_over":binary.LittleEndian.PutUint16(d.Data[slot+11:slot+13],256)
 case "source_length_exceeds_extent":binary.LittleEndian.PutUint16(d.Data[slot+11:slot+13],257)
 case "free_overlap":sys[225]=0;sys[226]=1
 case "no_space":sys[229]=0;sys[230]=0
 case "directory_full":for i:=count;i<128;i++{copy(d.Data[i*16:i*16+9],fmt.Sprintf("F%07dD",i))};sys[228]=128
 case "directory_slot_15","directory_slot_16","directory_slot_127":end:=15;if control=="directory_slot_16"{end=16};if control=="directory_slot_127"{end=127};for i:=count;i<end;i++{copy(d.Data[i*16:i*16+9],fmt.Sprintf("F%07dD",i))};sys[228]=byte(end)
 case "last_sector":sys[225]=15;sys[226]=159;sys[229]=1;sys[230]=0
 case "full_disk":sys[225]=0;sys[226]=160;sys[229]=0;sys[230]=0
 case "bad_full_sentinel":sys[225]=1;sys[226]=160;sys[229]=0;sys[230]=0
 case "space_exact","space_one_short":free:=215;if control=="space_one_short"{free=214};pos:=2560-free;sys[225]=byte(pos%16);sys[226]=byte(pos/16);binary.LittleEndian.PutUint16(sys[229:231],uint16(free))
 default:panic("unknown echo disk control")
 }
 putLinearData(d,8,sys)
}
