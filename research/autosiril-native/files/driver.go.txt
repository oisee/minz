// Research OS adapter: actual files and record I/O only, no MIDI/music logic.
type diskOS struct {
	dir            string
	read           func(int64, int) []byte
	write          func(int64, byte)
	dma            int64
	handles        map[uint16]*os.File
	calls          map[byte]int
	failFunction   byte
	failOccurrence int
}

func (d *diskOS) invoke(fn byte, de uint16) byte {
	d.calls[fn]++
	if fn == d.failFunction && d.calls[fn] == d.failOccurrence {
		if fn == 16 {
			if f := d.handles[de]; f != nil {
				f.Close()
				delete(d.handles, de)
			}
		}
		if fn == 15 || fn == 22 || fn == 16 {
			return 255
		}
		return 2
	}
	addr := int64(de)
	record := func() int64 {
		return (int64(d.read(addr+14, 1)[0])*32+int64(d.read(addr+12, 1)[0]))*128 + int64(d.read(addr+32, 1)[0])
	}
	advance := func(n int64) {
		d.write(addr+32, byte(n%128))
		d.write(addr+12, byte(n/128%32))
		d.write(addr+14, byte(n/4096))
	}
	switch fn {
	case 26:
		d.dma = addr
		return 0
	case 15, 22:
		if d.handles[de] != nil {
			panic("FCB reopened without close")
		}
		b := d.read(addr+1, 11)
		name := strings.TrimSpace(string(b[:8])) + "." + strings.TrimSpace(string(b[8:]))
		if name != "INPUT.MID" && name != "NOTES.BIN" {
			panic("unexpected OS filename: " + name)
		}
		var f *os.File
		var e error
		if fn == 15 {
			f, e = os.Open(filepath.Join(d.dir, name))
		} else {
			f, e = os.Create(filepath.Join(d.dir, name))
		}
		if e != nil {
			return 255
		}
		d.handles[de] = f
		advance(0)
		return 0
	case 16:
		f := d.handles[de]
		if f == nil {
			return 255
		}
		delete(d.handles, de)
		if f.Close() != nil {
			return 255
		}
		return 0
	case 20, 21:
		f := d.handles[de]
		if f == nil {
			return 255
		}
		if d.read(addr+32, 1)[0] > 127 || d.read(addr+12, 1)[0] > 31 || d.read(addr+14, 1)[0] > 63 {
			return 2
		}
		r := record()
		if fn == 20 {
			b := bytes.Repeat([]byte{26}, 128)
			n, e := f.ReadAt(b, r*128)
			if e != nil && e != io.EOF {
				return 2
			}
			if n == 0 {
				return 1
			}
			for i, v := range b {
				d.write(d.dma+int64(i), v)
			}
		} else {
			b := d.read(d.dma, 128)
			n, e := f.WriteAt(b, r*128)
			if e != nil || n != 128 {
				return 2
			}
		}
		advance(r + 1)
		return 0
	default:
		panic(fmt.Sprintf("unadmitted BDOS %d", fn))
	}
}
func (d *diskOS) close() {
	for _, f := range d.handles {
		f.Close()
	}
}

// This host-side decoder is an oracle for the wire representation; it does not
// produce musical records for the target. A target reader is the next stage.
func checkWire(data []byte, want []note, sources uint16) {
	header := []byte{'A', 'S', 'I', 'R', 1, 1, 6, 0, byte(sources), byte(sources >> 8), 0, 0}
	if len(data) < len(header) || !bytes.Equal(data[:12], header) {
		panic("wire header differs")
	}
	at := 12
	for i, n := range want {
		b := []byte{1, byte(n.Start), byte(n.Start >> 8), byte(n.Off), byte(n.Off >> 8), n.Pitch, n.Source}
		if at+7 > len(data) || !bytes.Equal(data[at:at+7], b) {
			panic(fmt.Sprintf("wire record %d differs", i))
		}
		at += 7
	}
	n := uint32(len(want))
	end := []byte{255, byte(n), byte(n >> 8), byte(n >> 16), byte(n >> 24), 'E', 'N', 'D', '!'}
	if at+9 > len(data) || !bytes.Equal(data[at:at+9], end) {
		panic("wire logical END/count differs")
	}
	at += 9
	if len(data) != (at+127)/128*128 {
		panic("wire physical length differs")
	}
	for _, v := range data[at:] {
		if v != 0 {
			panic("wire physical padding differs")
		}
	}
}
func main() {
	source := flag.String("source", "", "composed target Nanz")
	root := flag.String("root", "..", "repository root")
	input := flag.String("input", "", "case matrix")
	music := flag.String("music", "", "fresh Crystal export (defaults to frozen fixture)")
	manifest := flag.String("manifest", "", "fresh Crystal manifest (defaults to frozen fixture)")
	out := flag.String("out", "", "outputs outside checkout")
	flag.Parse()
	if *out == "" || *source == "" {
		panic("source/out required")
	}
	check(os.MkdirAll(*out, 0755))
	fixtures := filepath.Join(*root, "research/autosiril-native/midi/fixtures")
	if *music == "" {
		*music = filepath.Join(fixtures, "music.lanz")
	}
	if *manifest == "" {
		*manifest = filepath.Join(fixtures, "music-manifest.json")
	}
	buildOne := func() *hir.Module {
		h, e := build(read(*source), read(*music), read(*manifest), filepath.Join(*root, "stdlib"))
		check(e)
		return h
	}
	raw := hir.LowerModule(buildOne())
	check(mir2.Verify(raw))
	opts := pipeline.DefaultOptions()
	opts.AssertMode = "none"
	opts.DisableVariantSearch = true
	compiled, e := pipeline.CompileHIRSteps(buildOne(), opts)
	check(e)
	// Standalone .COM bootstrap uses the Chronos profile's eight rows/beat.
	// Matrix engines enter disk_main directly with each fixture's own setting.
	assembly := "ORG 0x100\nLD SP,0xeffe\nLD HL,8\nLD (per_beat),HL\nLD HL,0\nLD (per_beat+2),HL\nCALL disk_main\nHALT\n" + compiled.Assembly
	a, e := z80asm.NewAssembler().AssembleString(assembly)
	check(e)
	if len(a.Errors) != 0 {
		panic(a.Errors)
	}
	if origin+len(a.Binary)+64 >= floor {
		panic("image overlaps stack")
	}
	for name, b := range map[string][]byte{"program.a80": []byte(assembly), "program.com": a.Binary, "raw.mir": []byte(raw.Dump()), "final.mir": []byte(compiled.MIR2Module.Dump())} {
		check(os.WriteFile(filepath.Join(*out, name), b, 0644))
	}
	type fileCase struct {
		testCase
		OSFault struct {
			Function   byte `json:"function"`
			Occurrence int  `json:"occurrence"`
		} `json:"os_fault"`
	}
	var matrix struct {
		Cases []fileCase `json:"cases"`
	}
	check(json.Unmarshal(read(*input), &matrix))
	if len(matrix.Cases) == 0 {
		panic("empty matrix")
	}
	summaries := []map[string]any{}
	for _, tc := range matrix.Cases {
		for engineIndex, engine := range []string{"raw-mir2", "final-mir2", "z80"} {
			dir := filepath.Join(*out, tc.Name, engine)
			check(os.MkdirAll(dir, 0755))
			check(os.WriteFile(filepath.Join(dir, "INPUT.MID"), tc.MIDI, 0644))
			d := &diskOS{dir: dir, handles: map[uint16]*os.File{}, calls: map[byte]int{}, failFunction: tc.OSFault.Function, failOccurrence: tc.OSFault.Occurrence}
			var get func(string, int, int) []byte
			var targetReturn uint16
			cycles, stack := 0, 0
			if engineIndex < 2 {
				m := raw
				if engineIndex == 1 {
					m = compiled.MIR2Module
				}
				v := mir2.NewVM(m)
				v.MaxMemory = 65536
				v.MaxSteps = 1000000000
				get = func(name string, off, n int) []byte { return v.ReadHeap(v.GlobalAddr(name)+int64(off), n) }
				d.read = v.ReadHeap
				d.write = v.WriteHeap
				for bank := 0; bank < 3; bank++ {
					for i := 4092; i < 4096; i++ {
						v.WriteHeap(v.GlobalAddr("note_bytes")+int64(bank*4096+i), 0xda)
					}
				}
				host := func(args []mir2.Value) ([]mir2.Value, error) {
					return []mir2.Value{{I: int64(d.invoke(byte(args[0].I), uint16(args[1].I)))}}, nil
				}
				v.Hosts["bdos8"] = host
				v.Hosts["@bdos8"] = host
				for i := 0; i < 4; i++ {
					v.WriteHeap(v.GlobalAddr("per_beat")+int64(i), byte(tc.Beat>>uint(8*i)))
				}
				r, e := v.Call("disk_main", nil)
				check(e)
				if len(r) != 1 {
					panic("invalid VM return")
				}
				targetReturn = uint16(r[0].I)
			} else {
				z := emulator.NewRemogattoZ80WithScreen()
				z.SetROMEnd(0)
				p := emulator.NewProfiler()
				z.SetProfiler(p)
				mem := z.Memory()
				for i := range mem {
					mem[i] = 0xa5
				}
				copy(mem[origin:], a.Binary)
				d.read = func(at int64, n int) []byte { return mem[int(at) : int(at)+n] }
				d.write = func(at int64, b byte) { z.WriteMemory(uint16(at), b) }
				get = func(name string, off, n int) []byte { at := a.Symbols[name] + off; return mem[at : at+n] }
				for bank := 0; bank < 3; bank++ {
					for i := 4092; i < 4096; i++ {
						mem[a.Symbols["note_bytes"]+bank*4096+i] = 0xda
					}
				}
				for i := 0; i < 4; i++ {
					mem[a.Symbols["per_beat"]+i] = byte(tc.Beat >> uint(8*i))
				}
				mem[initialSP] = 0
				mem[initialSP+1] = 0xf1
				z.SetRegisters(emulator.Registers{PC: uint16(a.Symbols["disk_main"]), SP: initialSP})
				minSP := uint16(initialSP)
				for steps := 0; z.GetPC() != returnPC; steps++ {
					if steps >= 250000000 {
						panic("native instruction limit")
					}
					if z.GetPC() != 5 && (int(z.GetPC()) < origin || int(z.GetPC()) >= origin+len(a.Binary)) {
						panic("native PC outside admitted image/BDOS")
					}
					if z.GetPC() == 5 {
						r := z.GetRegisters()
						answer := d.invoke(byte(r.BC), r.DE)
						r.A = answer
						r.HL = 0
						r.PC = binary.LittleEndian.Uint16(mem[r.SP : r.SP+2])
						r.SP += 2
						z.SetRegisters(r)
					} else {
						z.Step()
					}
					if z.GetSP() < minSP {
						minSP = z.GetSP()
					}
					if z.GetSP() < floor {
						panic("native stack floor")
					}
				}
				if z.GetSP() != initialSP+2 {
					panic("native return stack")
				}
				targetReturn = z.GetRegisters().HL
				allowed := make([]bool, 65536)
				for _, g := range compiled.MIR2Module.Globals {
					if g.At != nil {
						panic("unexpected absolute global")
					}
					at, ok := a.Symbols[g.Name]
					if !ok {
						panic("missing global")
					}
					for i := 0; i < mir2.ByteWidth(g.Ty); i++ {
						allowed[at+i] = true
					}
				}
				for _, s := range backendSpills(compiled.Assembly, a.Symbols, a.Binary) {
					for i := 0; i < s.Bytes; i++ {
						allowed[s.Address+i] = true
					}
				}
				for i := floor; i < initialSP; i++ {
					allowed[i] = true
				}
				for at, w := range p.WriteCount {
					if w != 0 && !allowed[at] {
						panic(fmt.Sprintf("native write outside storage %04x", at))
					}
					if allowed[at] && at < floor && p.ExecCount[at] != 0 {
						panic("executed data")
					}
				}
				if len(p.IORead) != 0 || len(p.IOWrite) != 0 {
					panic("unexpected port I/O")
				}
				for bank := 0; bank < 3; bank++ {
					for i := 4092; i < 4096; i++ {
						if p.WriteCount[a.Symbols["note_bytes"]+bank*4096+i] != 0 {
							panic("native wrote note arena padding")
						}
					}
				}
				for i := 0; i < 4; i++ {
					if p.WriteCount[a.Symbols["per_beat"]+i] != 0 {
						panic("native wrote immutable beat input")
					}
				}
				for _, s := range [][2]int{{origin - 64, origin}, {origin + len(a.Binary), origin + len(a.Binary) + 64}, {floor - 64, floor}} {
					for i := s[0]; i < s[1]; i++ {
						if mem[i] != 0xa5 {
							panic("native canary")
						}
					}
				}
				cycles = z.GetCycles()
				stack = int(initialSP - minSP)
			}
			if len(d.handles) != 0 {
				d.close()
				panic("target leaked open file")
			}
			got := collect(engine, get)
			for bank := 0; bank < 3; bank++ {
				for i := 4092; i < 4096; i++ {
					if get("note_bytes", bank*4096+i, 1)[0] != 0xda {
						panic("target mutated note arena padding")
					}
				}
			}
			if targetReturn != got.Fault || got.Fault != tc.Expected.Fault || got.Sources != tc.Expected.Sources || len(got.Notes) != len(tc.Expected.Notes) || got.Active != tc.Expected.Active {
				panic(fmt.Sprintf("%s/%s state differs: %+v", tc.Name, engine, got))
			}
			for i, n := range got.Notes {
				if n != tc.Expected.Notes[i] {
					panic(fmt.Sprintf("%s/%s note%d differs", tc.Name, engine, i))
				}
			}
			if tc.Expected.Peak != nil && got.Peak != *tc.Expected.Peak {
				panic("target peak differs")
			}
			if tc.Expected.Bytes != nil && got.Bytes != *tc.Expected.Bytes {
				panic(fmt.Sprintf("%s/%s accepted bytes %d != %d", tc.Name, engine, got.Bytes, *tc.Expected.Bytes))
			}
			if tc.Expected.Events != nil && got.Events != *tc.Expected.Events {
				panic("target event count differs")
			}
			wireHash := ""
			wireBytes := 0
			if got.Fault == 0 {
				b := read(filepath.Join(dir, "NOTES.BIN"))
				checkWire(b, tc.Expected.Notes, tc.Expected.Sources)
				wireHash = hash(b)
				wireBytes = len(b)
			} else if got.Fault < 901 || got.Fault > 903 {
				if _, e := os.Stat(filepath.Join(dir, "NOTES.BIN")); !os.IsNotExist(e) {
					panic("failed parser/input transport created output")
				}
			}
			summaries = append(summaries, map[string]any{"case": tc.Name, "engine": engine, "fault": got.Fault, "notes": len(got.Notes), "bytes": got.Bytes, "events": got.Events, "peak": got.Peak, "wire_sha256": wireHash, "wire_bytes": wireBytes, "bdos_calls": d.calls, "t_states": cycles, "stack_below_entry": stack})
			fmt.Printf("PASS %s/%s notes%d fault%d output%dB %dT stack%d\n", tc.Name, engine, len(got.Notes), got.Fault, wireBytes, cycles, stack)
		}
	}
	globals, spills := 0, 0
	for _, g := range compiled.MIR2Module.Globals {
		globals += mir2.ByteWidth(g.Ty)
	}
	for _, s := range backendSpills(compiled.Assembly, a.Symbols, a.Binary) {
		spills += s.Bytes
	}
	receipt := map[string]any{"protocol": "autosiril-file-pair-probe/v1", "source_sha256": hash(read(*source)), "input_sha256": hash(read(*input)), "binary_sha256": hash(a.Binary), "image_bytes": len(a.Binary), "module_globals_bytes": globals, "backend_spill_bytes": spills, "code_bytes": len(a.Binary) - globals - spills, "target_executions": len(summaries), "cases": summaries, "scope": "CP/M record adapter with real host files; raw/final MIR2 and native CALL5; independent mze execution still required; target file reader/flatten stage pending"}
	b, e := json.MarshalIndent(receipt, "", "  ")
	check(e)
	check(os.WriteFile(filepath.Join(*out, "receipt.json"), append(b, '\n'), 0644))
}
