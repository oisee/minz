const segmentBytes=65280
func linearData(d *formats.TRDFile,pos int)[]byte{t:=pos/16;return d.ReadSector(t%80,t/80,pos%16)}
func linearOffset(pos int)int{t:=pos/16;return ((t%80*2+t/80)*16+pos%16)*256}
func putLinearData(d *formats.TRDFile,pos int,b []byte){t:=pos/16;d.WriteSector(t%80,t/80,pos%16,b)}
// Segments of one logical stream: every non-final segment is exactly 255 full sectors.
func segments(d *formats.TRDFile,prefix string,digit int,ext byte)(data []byte,count int){
 for i:=0;i<10;i++{
  name:=[]byte(prefix);name[digit]=byte('0'+i)
  entry:=d.FindFile(strings.TrimRight(string(name)," "),ext);if entry==nil{break}
  b:=d.ReadFile(entry);if len(b)==0||len(b)>segmentBytes{panic("segment length")}
  if count>0&&len(data)%segmentBytes!=0{panic("short non-final segment")}
  data=append(data,b...);count++
 }
 return
}
