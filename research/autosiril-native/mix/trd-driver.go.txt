func linearData(d *formats.TRDFile,pos int)[]byte{t:=pos/16;return d.ReadSector(t%80,t/80,pos%16)}
func linearOffset(pos int)int{t:=pos/16;return ((t%80*2+t/80)*16+pos%16)*256}
func putLinearData(d *formats.TRDFile,pos int,b []byte){t:=pos/16;d.WriteSector(t%80,t/80,pos%16,b)}
func seedDisk(notes,wire []byte)*formats.TRDFile{
 d:=&formats.TRDFile{};pos:=16
 for i,data:=range [][]byte{notes,wire}{
  if len(data)>65280{panic("TRD input not representable")}
  slot:=i*16;name:="NOTES   N";if i==1{name="DELAYED D"};copy(d.Data[slot:slot+9],name)
  n:=(len(data)+255)/256;binary.LittleEndian.PutUint16(d.Data[slot+11:slot+13],uint16(len(data)));d.Data[slot+13]=byte(n);d.Data[slot+14]=byte(pos%16);d.Data[slot+15]=byte(pos/16)
  for j:=0;j<n;j++{b:=make([]byte,256);end:=(j+1)*256;if end>len(data){end=len(data)};copy(b,data[j*256:end]);putLinearData(d,pos+j,b)};pos+=n
 }

}
