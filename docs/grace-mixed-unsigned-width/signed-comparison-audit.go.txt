package main
import("fmt";"github.com/minz/minzc/pkg/nanz";"github.com/minz/minzc/pkg/hir";"github.com/minz/minzc/pkg/mir2";"github.com/minz/minzc/pkg/pipeline";"github.com/minz/minzc/pkg/z80asm";"github.com/minz/minzc/pkg/emulator")
func main(){src:=`global narrow:i16=65535
 global wide:i32=0
 global lr:u8
 global rl:u8
 global explicit_lr:u8
 fun compute(){lr=u8(narrow<wide)
 rl=u8(wide>narrow)
 explicit_lr=u8(i32(narrow)<wide)}`
 h,e:=nanz.Parse(src,"comparison-audit");if e!=nil{panic(e)};r:=hir.LowerModule(h);v:=mir2.NewVM(r);_,e=v.Call("compute",nil);fmt.Println("raw",v.ReadHeap(6,3),e);s,e:=pipeline.CompileHIRSteps(h,pipeline.Options{AssertMode:"none"});if e!=nil{panic(e)};v=mir2.NewVM(s.MIR2Module);_,e=v.Call("compute",nil);fmt.Println("final",v.ReadHeap(6,3),e);a,e:=z80asm.NewAssembler().AssembleString("ORG 0x8000\n"+s.Assembly);if e!=nil||len(a.Errors)>0{panic(e)};z:=emulator.NewRemogattoZ80();z.LoadMemory(0x8000,a.Binary);z.Memory()[0xeff0]=0;z.Memory()[0xeff1]=3;z.SetRegisters(emulator.Registers{PC:uint16(a.Symbols["compute"]),SP:0xeff0});for i:=0;i<10000&&z.GetPC()!=0x300;i++{z.Step()};fmt.Println("native",z.Memory()[a.Symbols["lr"]],z.Memory()[a.Symbols["rl"]],z.Memory()[a.Symbols["explicit_lr"]],"PCSP",z.GetPC(),z.GetSP());fmt.Println(s.Assembly)
}
