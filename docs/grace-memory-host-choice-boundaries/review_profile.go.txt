package main
import("encoding/json";"fmt";"os";"path/filepath";"github.com/minz/minzc/pkg/hir";"github.com/minz/minzc/pkg/nanz")
func must(e error){if e!=nil{panic(e)}}
func main(){source,e:=os.ReadFile(os.Args[1]);must(e);files,e:=filepath.Glob(filepath.Join(os.Args[2],"*.json"));must(e);for _,file:=range files{b,e:=os.ReadFile(file);must(e);var w hir.MemoryHostWorkload;must(json.Unmarshal(b,&w));m,e:=nanz.Parse(string(source),os.Args[1]);must(e);m.Target=hir.TargetCPM;run,err:=hir.RunMemoryHostLoopProfile(m,w);name:=filepath.Base(file);fmt.Printf("%s error=%v\n",name,err);if run==nil{fmt.Println("nil profile");continue};fmt.Printf(" completed%d failed%d\n",run.CompletedCalls,run.FailedCalls);for i,s:=range run.Sites.Sites{if s.Function=="region_match"||s.Function=="intern_region"{fmt.Printf(" %s %s entries%d body%d\n",s.Function,s.HIRPath,run.Counts[i].Entries,run.Counts[i].BodyEntries)}};for i,inv:=range run.Invocations{fmt.Printf(" call%d error=%s failure=%+v events%d\n",i,inv.Observation.Error,inv.Observation.Failure,len(inv.Observation.Events))};encoded,e:=json.MarshalIndent(run,"","  ");must(e);must(os.WriteFile(filepath.Join(os.Args[3],name),encoded,0644))}}
