package main
import("fmt";"os";"path/filepath";"strings";"github.com/minz/minzc/pkg/nanz";"github.com/minz/minzc/pkg/pipeline";"github.com/minz/minzc/pkg/z80asm")
func main(){for _,path:=range os.Args[2:]{b,e:=os.ReadFile(path);if e!=nil{panic(e)};name:=strings.TrimSuffix(filepath.Base(path),".nanz");h,e:=nanz.Parse(string(b),name);if e!=nil{panic(e)};opts:=pipeline.DefaultOptions();opts.AssertMode="none";a,e:=pipeline.CompileHIRWithOptions(h,opts);if e!=nil{panic(e)};p:=filepath.Join(os.Args[1],name);os.MkdirAll(p,0755);os.WriteFile(filepath.Join(p,"showcase-route.a80"),[]byte(a),0644);r,e:=z80asm.NewAssembler().AssembleString(a);text:=fmt.Sprintf("assemble error: %v\n",e);if r!=nil{for _,err:=range r.Errors{text+=fmt.Sprintln(err)}};os.WriteFile(filepath.Join(p,"assemble.txt"),[]byte(text),0644);fmt.Print(name,"\n",text)}}
