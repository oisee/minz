package pipeline

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"github.com/minz/minzc/pkg/hir"
	"github.com/minz/minzc/pkg/mir2"
	"github.com/minz/minzc/pkg/nanz"
	"os"
	"path/filepath"
	"sort"
	"strings"
	"testing"
)

func TestTokenizerBoundedABIRefinement(t *testing.T) {
	const report = "../../../reports/pfcco-tokenizer-objective"
	const path = "../../../examples/nanz/self_tokenizer.nanz"
	dir := os.Getenv("MINZ_TOKENIZER_REFINEMENT_EXPORT")
	if dir == "" {
		t.Skip("bounded experiment opt-in")
	}
	src, e := os.ReadFile(path)
	if e != nil {
		t.Fatal(e)
	}
	var freeze struct {
		Source string            `json:"source_sha256"`
		Inputs map[string]string `json:"inputs_sha256"`
	}
	b, e := os.ReadFile(report + "/frozen/freeze.json")
	if e != nil {
		t.Fatal(e)
	}
	if e = json.Unmarshal(b, &freeze); e != nil {
		t.Fatal(e)
	}
	if fmt.Sprintf("%x", sha256.Sum256(src)) != freeze.Source {
		t.Fatal("source changed")
	}
	load := func(name string) []string {
		b, e := os.ReadFile(report + "/frozen/" + name + ".json")
		if e != nil {
			t.Fatal(e)
		}
		if fmt.Sprintf("%x", sha256.Sum256(b)) != freeze.Inputs[name+".json"] {
			t.Fatal("workload changed")
		}
		var v []string
		if e = json.Unmarshal(b, &v); e != nil {
			t.Fatal(e)
		}
		return v
	}
	train, held := load("training"), load("heldout")
	factory := func(strategy string) (*hir.Module, Steps) {
		h, e := nanz.Parse(string(src), path)
		if e != nil {
			t.Fatal(e)
		}
		s, e := CompileHIRSteps(h, Options{ContractOpt: true, ContractStrategy: strategy, OptSize: true, AssertMode: "none"})
		if e != nil {
			t.Fatal(e)
		}
		return h, s
	}
	_, greedy := factory("")
	_, pbqp := factory("pbqp")
	// Groups are structural position/type differences; no tokenizer names.
	groups := map[string]map[string][]int{}
	for _, f := range pbqp.MIR2Module.Funcs {
		g := greedy.MIR2Module.FuncByName(f.Name)
		if g == nil || len(g.Contract.Params) != len(f.Contract.Params) {
			t.Fatal("candidate shape drift")
		}
		for i, p := range f.Contract.Params {
			if p.Ty.String() != g.Contract.Params[i].Ty.String() {
				t.Fatal("type drift")
			}
			if p.Class != g.Contract.Params[i].Class {
				if f.Attrs.IsExtern || f.Attrs.FixedIndirectABI || f.Attrs.FallibleCY || f.Attrs.HasAsm {
					t.Fatal("fixed ABI difference")
				}
				key := fmt.Sprintf("slot%d-%s", i, p.Ty.String())
				if groups[key] == nil {
					groups[key] = map[string][]int{}
				}
				groups[key][f.Name] = append(groups[key][f.Name], i)
			}
		}
	}
	keys := []string{}
	for k := range groups {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	if len(keys) > 4 {
		t.Fatal("bounded domain exceeds four groups")
	}
	type candidate struct {
		Name     string
		T, Bytes int
		Steps    Steps       `json:"-"`
		HIR      *hir.Module `json:"-"`
	}
	var rows []candidate
	names := append([]string{"greedy", "pbqp"}, keys...)
	for _, name := range names {
		strategy := "pbqp"
		if name == "greedy" {
			strategy = ""
		}
		h, s := factory(strategy)
		if name != "greedy" && name != "pbqp" {
			proposal := &contractRefinement{Classes: map[string]map[int]mir2.RegClass{}}
			for fn, positions := range groups[name] {
				proposal.Classes[fn] = map[int]mir2.RegClass{}
				for _, i := range positions {
					proposal.Classes[fn][i] = greedy.MIR2Module.FuncByName(fn).Contract.Params[i].Class
				}
			}
			opts := Options{ContractOpt: true, ContractStrategy: "pbqp", OptSize: true, AssertMode: "none", contractRefinement: proposal}
			h, e = nanz.Parse(string(src), path)
			if e != nil {
				t.Fatal(e)
			}
			_, err := CompileHIRSteps(h, opts)
			const prefix = "experimental contract refinement context changed: "
			if err == nil || !strings.HasPrefix(err.Error(), prefix) {
				t.Fatal("missing canonical refinement binding", err)
			}
			proposal.ExpectedSHA256 = strings.TrimPrefix(err.Error(), prefix)
			h, e = nanz.Parse(string(src), path)
			if e != nil {
				t.Fatal(e)
			}
			s, e = CompileHIRSteps(h, opts)
			if e != nil {
				t.Fatal(e)
			}
		}
		out := filepath.Join(dir, name, "training")
		if e = os.MkdirAll(out, 0755); e != nil {
			t.Fatal(e)
		}
		cost, size := runTokenizerABICandidate(t, h, s, train, out, name)
		rows = append(rows, candidate{Name: name, T: cost, Bytes: size, Steps: s, HIR: h})
		t.Logf("candidate%s entiretrain %dT/%dB", name, cost, size)
	}
	// A candidate failure above aborts before either winner or heldout runs.
	for _, goal := range []string{"speed", "size"} {
		best := 0
		for i := 1; i < len(rows); i++ {
			if (goal == "speed" && rows[i].T < rows[best].T) || (goal == "size" && rows[i].Bytes < rows[best].Bytes) {
				best = i
			}
		}
		selected := rows[best]
		asm := selected.Steps.Assembly
		out := filepath.Join(dir, goal+"-heldout")
		if e = os.MkdirAll(out, 0755); e != nil {
			t.Fatal(e)
		}
		cost, size := runTokenizerABICandidate(t, selected.HIR, selected.Steps, held, out, selected.Name)
		if selected.Steps.Assembly != asm || size != selected.Bytes {
			t.Fatal("selected executable changed")
		}
		t.Logf("%s selected%s fixedBIN heldout %dT/%dB", goal, selected.Name, cost, size)
	}
	b, e = json.MarshalIndent(rows, "", "  ")
	if e != nil {
		t.Fatal(e)
	}
	if e = os.WriteFile(filepath.Join(dir, "candidates.json"), b, 0644); e != nil {
		t.Fatal(e)
	}
}
